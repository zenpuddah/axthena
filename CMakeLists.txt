cmake_minimum_required(VERSION 3.24)
project(axthena VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
find_package(SQLite3 REQUIRED)
find_package(CURL REQUIRED)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib URL https://codeload.github.com/yhirose/cpp-httplib/tar.gz/refs/tags/v0.61.0 URL_HASH SHA256=f03bdb0bee4911d6db0335bc103f0f441ae70c527e993c1f4b3f177586b042d3)
FetchContent_MakeAvailable(httplib)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama URL https://github.com/ggml-org/llama.cpp/archive/2a667032dc788c3a071d538e0b572e45a3f12731.tar.gz URL_HASH SHA256=59ba9d6660156f6f70d82d8f0f9de8d2aa4b3df4456dd5cb40502ac5f176e48f)
FetchContent_MakeAvailable(llama)
add_library(axthena_core src/core.cpp src/store.cpp src/neo4j.cpp src/legacy_import.cpp src/model.cpp src/embedding.cpp src/pipeline.cpp src/semantics.cpp src/query.cpp src/collection.cpp src/ingestion.cpp src/publisher_text.cpp src/domains.cpp src/investigation.cpp src/workspace.cpp)
target_include_directories(axthena_core PUBLIC include)
if(TARGET SQLite3::SQLite3)
  set(AXTHENA_SQLITE SQLite3::SQLite3)
else()
  set(AXTHENA_SQLITE SQLite::SQLite3)
endif()
target_link_libraries(axthena_core PUBLIC nlohmann_json::nlohmann_json ${AXTHENA_SQLITE} CURL::libcurl llama httplib::httplib)
target_compile_options(axthena_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(axthena_core PUBLIC AXTHENA_COMPONENT="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(axthena src/main.cpp)
target_link_libraries(axthena PRIVATE axthena_core)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT OR CMAKE_INSTALL_PREFIX STREQUAL "/usr/local")
  set(CMAKE_INSTALL_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/install" CACHE PATH "Local Axthena runtime" FORCE)
endif()
install(TARGETS axthena RUNTIME DESTINATION bin COMPONENT AxthenaRuntime)
install(DIRECTORY config prompts web examples src include DESTINATION share/axthena COMPONENT AxthenaRuntime)
install(FILES tools/ingest.py tools/ingestors.py DESTINATION share/axthena/tools COMPONENT AxthenaRuntime)
install(DIRECTORY workspace/dist DESTINATION share/axthena/workspace COMPONENT AxthenaRuntime OPTIONAL)
install(DIRECTORY docs DESTINATION share/axthena COMPONENT AxthenaRuntime)
install(FILES CMakeLists.txt CMakePresets.json README.md DESTINATION share/axthena COMPONENT AxthenaRuntime)
install(DIRECTORY tests/fixtures DESTINATION share/axthena/tests COMPONENT AxthenaRuntime)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_test(NAME ingestion_orchestration COMMAND ${CMAKE_COMMAND} -E env AXTHENA_NATIVE=$<TARGET_FILE:axthena> ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ingestion_tests.py -v)
  add_executable(axthena_tests tests/native_tests.cpp)
  target_link_libraries(axthena_tests PRIVATE axthena_core)
  add_test(NAME native_contracts COMMAND axthena_tests)
  set_tests_properties(native_contracts PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_executable(axthena_semantics_tests tests/semantics_tests.cpp)
  target_link_libraries(axthena_semantics_tests PRIVATE axthena_core)
  add_test(NAME semantic_contracts COMMAND axthena_semantics_tests)
  add_executable(axthena_recovery_tests tests/recovery_tests.cpp)
  target_link_libraries(axthena_recovery_tests PRIVATE axthena_core)
  add_test(NAME transport_recovery COMMAND axthena_recovery_tests)
  add_executable(axthena_pipeline_tests tests/pipeline_tests.cpp)
  target_link_libraries(axthena_pipeline_tests PRIVATE axthena_core)
  add_test(NAME native_insertion COMMAND axthena_pipeline_tests)
  set_tests_properties(native_insertion PROPERTIES LABELS "encoder")
  add_executable(axthena_collection_tests tests/collection_tests.cpp)
  target_link_libraries(axthena_collection_tests PRIVATE axthena_core)
  add_test(NAME public_collection COMMAND axthena_collection_tests)
  add_executable(axthena_neo4j_tests tests/neo4j_tests.cpp)
  target_link_libraries(axthena_neo4j_tests PRIVATE axthena_core)
  add_test(NAME neo4j_integration COMMAND axthena_neo4j_tests)
  add_executable(axthena_investigation_tests tests/investigation_tests.cpp)
  target_link_libraries(axthena_investigation_tests PRIVATE axthena_core)
  add_test(NAME investigation_contracts COMMAND axthena_investigation_tests)
endif()
